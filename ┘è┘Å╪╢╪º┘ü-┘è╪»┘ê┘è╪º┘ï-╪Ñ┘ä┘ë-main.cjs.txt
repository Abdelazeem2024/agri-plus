// ── القطعة 2: أضفها بجوار بقية الـ require في أول الملف
// (بعد سطر استيراد license.cjs إن وُجد، أو في أي مكان مناسب أعلى الملف)
const aiVoiceMod = require('./aiVoice.cjs');


// ── القطعة 3 (مهمة أمنياً): أضف هذا كأول سطر داخل app.whenReady().then(() => { ... })
// قبل أي كود آخر بداخلها — يسمح صراحة فقط بإذن الميكروفون (اللازم للمساعد
// الصوتي) ويرفض أي إذن آخر لم يطلبه البرنامج أصلاً
  session.defaultSession.setPermissionRequestHandler((webContents, permission, callback) => {
    callback(permission === 'media');
  });


// ── القطعة 4: أضف هذه المعالجات في أي مكان بعد تعريف mainWindow

ipcMain.handle('ai-voice-status', () => {
  try {
    return aiVoiceMod.getStatus();
  } catch (e) {
    return { modelReady: false, binaryReady: false, voiceDir: '', modelSizeApprox: 0, error: e.message };
  }
});

ipcMain.handle('ai-voice-download-model', async (event) => {
  try {
    const result = await aiVoiceMod.downloadModel((received, total) => {
      if (mainWindow && !mainWindow.isDestroyed()) {
        mainWindow.webContents.send('ai-voice-download-progress', received, total);
      }
    });
    return result;
  } catch (e) {
    return { success: false, error: e.message };
  }
});

ipcMain.handle('ai-transcribe-audio', async (_event, wavBase64) => {
  // wavBase64: ملف WAV (16kHz/أحادي) جاهز من src/lib/audioRecorder.ts في الواجهة
  return aiVoiceMod.transcribeAudio(wavBase64);
});

ipcMain.handle('ai-voice-available', () => {
  try {
    return aiVoiceMod.isVoiceReady();
  } catch {
    return false;
  }
});
